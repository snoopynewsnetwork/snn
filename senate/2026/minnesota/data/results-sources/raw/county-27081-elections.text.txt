Election & Voting Information from the
Minnesota Secretary of State's Office:                

Absentee Voting is held at the Lincoln County Courthouse (Auditor-Treasurer Office).Hours are as follows:Monday to Friday 8:30am to 4:30pmSaturday, October 24th 9am to 3pmTuesday, October 27th till 7pmSaturday, October 31st 9am to 3pmSunday, November 1st 9am to 3pmMonday, November 2nd till 5pm
The County Auditor-Treasurer is designated by the Secretary of State as Register of Voters and chief custodian of official voter registration and election records. The County Auditor-Treasurer is responsible for conducting the primary and general election during the even numbered years and also any special countywide elections. The County Auditor-Treasurer is responsible for training election judges and officials, printing ballots, acting as filing officer for county races, distributing absentee ballots to those who will be unable to vote on election day, tabulating the election results from the 20 precincts in the county. The County Auditor-Treasurer is also a member of the Canvassing Board. The Canvassing Board meets shortly after the election to review for accuracy the results of the election and to declare a winner.

	$(document).ready(function(){
	    var $category = $('.faq-category'),
	        $questionsAndAnswers = $('.faq-questions,.faq-answer'),
	        $filterCount = $(".faq-filter-count"),
	        $faqItem = $(".faq-item");

	    // remove inline scripts on load so not caught in filter
	    $(".faq-container script").remove();

	    var $filter = $("#faq-filter").keyup(function(){
	        $category.removeClass("has-search-result");

	        // Retrieve the input field text and reset the count to zero
	        var filter = $(this).val(),
	            count = 0;

	        // Loop through the questions list
	        $faqItem.each(function(){
	            var $this = $(this);
	            // If the questions item does not contain the text phrase fade it out
	            if ($this.text().search(new RegExp(filter, "i")) < 0) {
	                $this.fadeOut();
						 $this.find('.faq-question>a').attr('style','');
						 $('.active-question').removeClass('active-question');
	            // Show the questions item if the phrase matches and increase the count by 1
	            } else {
	                $this.fadeIn().closest(".faq-category").addClass("has-search-result");
						 $this.find('.faq-question>a').addClass('active-question').css('pointer-events','none');
	                count++;
	            }
	        });

	        if( filter !== '' ) {
	            $(".faq-category:not(.has-search-result)").fadeOut();
	            $(".faq-category.has-search-result").fadeIn();

	            // search should auto expand categories and answers when searching
	            $questionsAndAnswers.css({"display":"block", "visibility": "visible", "height": "auto"});
	            $filterCount.text(count + " items found. Please see below for details.");

	        } else {
	            // remove inline styles to revert classes back
	            $category.fadeIn();
	            $questionsAndAnswers.removeAttr("style");
	            $filterCount.text(" ");
					$('.active-question').removeClass('active-question');
					$('.faq-item .faq-question>a').attr('style','');
	        }
	    });

	    $filter.submit(function(e){
	        return false;
	    });

		// //location hashing
		$('.faq-question-header').on('click keypress', function(e) {
			let url = window.location.href.split("#")[0];
			console.log(url+this.dataset.target);
			history.pushState({}, '',url+this.dataset.target);
		});

		$('.faq-header').on('click keypress', function(e) {
			let url = window.location.href.split("#")[0];
			console.log(url+this.dataset.target);
			history.pushState({}, '',url+this.dataset.target);
		});

	    // disable toggling when searching
	    function ifSearchingDoNothing() {
	        if( $filter.val() !== '' ) {
	            return false;
	        }
	    }
	    $questionsAndAnswers.on("hide.bs.collapse show.bs.collapse", ifSearchingDoNothing);
		$questionsAndAnswers.on("hide.bs.collapse",function(event){
			$(event.target).siblings('.faq-question').find('a[data-toggle]').removeClass('active-question');
		});

		$questionsAndAnswers.on("show.bs.collapse",function(event){
			$(event.target).siblings('.faq-question').find('a[data-toggle]').addClass('active-question');
		});

		setTimeout(function() {
			if (window.location.hash != "") {
				var tempHash = window.location.hash;
				if (document.querySelector(tempHash) === null) { return; }
				if (tempHash.indexOf("#collapse") === 0) {
					var parent = document.querySelector(tempHash).parentElement.parentElement.parentElement;
					if (!parent.querySelector(".faq-questions").classList.contains('show')) {
						parent.querySelector("a.faq-header").click();
					}
					var el = document.querySelector(tempHash).parentElement.querySelector("a.faq-question-header");
					el.click();
					if ($) {
						$('html,body').animate({
							scrollTop: $(el).offset().top - 150
						}, 1000)
					}
				} else if (tempHash.indexOf("#faq-questions-") === 0) {
					var el = document.querySelector("a[data-target='"+tempHash+"']");
					if (!el.parentElement.querySelector(".faq-questions").classList.contains('show')) {
						el.click();
					}
					if ($) {
						$('html,body').animate({
							scrollTop: $(el).offset().top - 150
						}, 1000)
					}
				}
			}
		},100);

	});
