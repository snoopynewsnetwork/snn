document.addEventListener('DOMContentLoaded', function () {
	const $ = jQuery;

	const $input = $('#autocompleteme');

	const $clearBtn = $('.clear-search');

	const $searchPopoverText = $('.searchPopover-text');
	const $searchHelp = $('.search-help');
	const $popularPages = $('.popular-pages-search');
	const modal = document.getElementById('searchModal');

	$.widget("custom.noAutoCloseAutocomplete", $.ui.autocomplete, {
		_close: function () {},
		_renderItem: function (ul, item) {
			return $("<li>")
				.append("<div>" + item.label + "</div>")
				.appendTo(ul);
		}
	});

	const allPages = [];
	
	async function preloadSitemap() {
		if (allPages.length) { return; }
		
		fetch('_assets_/sitemap.xml').then(async function(res) {
			if (!res.ok) { return; }
			
			var xmlString = await res.text();
			
			var xmlParser = new DOMParser();
			var resDoc = xmlParser.parseFromString(xmlString, "application/xml");
			
			resDoc.querySelectorAll("url").forEach(function(el) {
				
				var lab = el.querySelector("title").textContent;
				var val = el.querySelector("loc").textContent;
				var link = el.querySelector("loc").textContent;
				
				allPages.push({label: lab, value: val, link: link});
			});
		});
	}

	function updateSearchUI(val) {
		if (val.length > 0) {
			$('.all-search-buttons .btn .search-query-text').text(val);
			if (!$searchButtonsWrap.is(':visible')) {
				$searchButtonsWrap.stop(true, true).slideDown(300);
			}
			$('.all-site-search-btn').attr('href', 'search.php?q=' + encodeURIComponent(val));
			$('.all-doc-search-btn').attr('href', 'search.php?q=' + encodeURIComponent('filetype:doc OR filetype:pdf OR filetype:docx ' + val));
			$popularPages.addClass('border-top');
		} else {
			$searchButtonsWrap.stop(true, true).slideUp(300);
			$searchHelp.hide(300);
			$noResults.hide(300);
			$popularPages.removeClass('border-top');
			$searchPopoverText.text("Or try searching...");
			$results.removeClass('open').hide(300);
		}
	}

	function initAutocomplete() {
		$input.noAutoCloseAutocomplete({
			appendTo: "#searchResults",
			minLength: 1,
			delay: 300,
			html: false,
			source: function (request, response) {
				const term = $.trim(request.term).toLowerCase();
				const words = term.split(/\s+/).filter(Boolean);

				$noResults.hide(300);
				$spinner.show();

				const matches = allPages.filter(item => {
					const value = item.value.toLowerCase();
					return words.every(w => value.includes(w));
				});

				// No highlighting â return label exactly as-is
				const cleaned = matches.map(item => ({
					label: item.label,
					value: item.value,
					link: item.link
				}));

				setTimeout(() => {
					$spinner.fadeOut(1000);

					if (!term || cleaned.length === 0) {
						$input.noAutoCloseAutocomplete('close');
						$('#searchResults ul').empty();
						$noResults.show(300);
						$searchHelp.hide(300);
						$results.removeClass('open').hide(300);
						$searchPopoverText.text("Or try searching...");
					} else {
						$noResults.hide(300);
						$searchHelp.show(300);
						$results.addClass('open').show(300);
						$searchPopoverText.text("If none of the results above are helpful, you can try searching...");
					}

					response(cleaned);
				}, 300);
			},
			select: function (event, ui) {
				window.location.href = ui.item.link;
			},
			open: function () {},
			close: function () {
				$results.removeClass('open').hide(300);
				$searchHelp.hide(300);
				$searchPopoverText.text("Or try searching...");
			}
		});
	}

	initAutocomplete();

function clickLanguage(langText, maxTries, intervalMs) {
	langText = (langText || '').trim();
	maxTries = maxTries || 30;
	intervalMs = intervalMs || 200;
	var tries = 0;
	var timer = setInterval(function () {
		var $frame = $("iframe.skiptranslate");
		if ($frame.length) {
			var doc = $frame.contents();
			var $target = doc.find('span.text').filter(function () {
				return $(this).text().trim() === langText;
			}).closest('a');
			if (!$target.length) {
				$target = doc.find('span.text').filter(function () {
					return $(this).text().trim() === 'English';
				}).closest('a');
			}
			if ($target.length) {
				$target.get(0).click();
				clearInterval(timer);
				return;
			}
		}
		tries++;
		if (tries >= maxTries) clearInterval(timer);
	}, intervalMs);
}

function getCookie(cname) {
	let name = cname + "=";
	let ca = document.cookie.split(';');
	for(let i = 0; i < ca.length; i++) {
		let c = ca[i];
		while (c.charAt(0) == ' ') c = c.substring(1);
		if (c.indexOf(name) == 0) return c.substring(name.length, c.length);
	}
	return "";
}

function sanitizeFlag(code) {
	var v = (code || "").toLowerCase().trim();
	if (!v || v === "undefined" || v === "null") return "";
	if (!/^[a-z]{2}$/.test(v)) return "";
	return v;
}

function applyActiveFromCookie() {
	var code = sanitizeFlag(getCookie('lang')) || sanitizeFlag(getCookie('code')) || 'us';
	var $btn = $('.translation-links button').filter(function(){
		var dataCode = ($(this).data('code') || '').toString().toLowerCase();
		if (dataCode && dataCode === code) return true;
		var cls = ($(this).find('span[class*="fi-"]').attr('class') || '');
		var m = cls.match(/\bfi-([a-z]{2})\b/i);
		return m && m[1].toLowerCase() === code;
	}).first();
	if (!$btn.length && code === 'us') $btn = $('.translation-links button[data-lang="English"]').first();
	$('.translation-links button').removeClass('active');
	if ($btn.length) $btn.addClass('active');
}

	function indexByIdOrRid(data){
		if (!data) return {};
		if (!Array.isArray(data)) return data;
		return data.reduce((m, item) => {
			if (!item) return m;
			const key = item.rid ?? item.id;

			return m;

	var parsedOptions = cRecord.options;
	if (typeof parsedOptions === 'string') {
		try { parsedOptions = JSON.parse(parsedOptions); } catch(e){ parsedOptions = {}; }
	}

	var imgURL = (cRecord.image || '').trim();

		m.img && m.img.classList.remove('d-none');

			m.img.style.backgroundImage = "url('" + imgURL + "')";

		}
	}

	// Title
	if (m.title) {
		var rawTitle = cRecord.title || cRecord.name || "";
		m.title.textContent = decodeHtmlEntities(rawTitle);
	}

	if (m.helpful) {
		var links = m.helpful.querySelectorAll("a");

function hasImg(url) {
	if (!url) return false;
	var s = String(url).trim();

		if (!scrollTarget || !progressBar) return;
	
		var maxScroll = scrollTarget.scrollHeight - scrollTarget.clientHeight;
		var current   = scrollTarget.scrollTop || 0;
	
		var pct = 0;
		if (maxScroll > 0) {
			pct = (current / maxScroll) * 100;
		}
		pct = Math.min(Math.max(pct, 0), 100);
	
		progressBar.style.width = pct + '%';
		if (progressOuter) {
			progressOuter.setAttribute('aria-valuenow', Math.round(pct));
		}
	}
	
	if (scrollTarget && progressBar) {
		scrollTarget.addEventListener('scroll', updateScrollProgress, { passive: true });
	
		modal.addEventListener('show.bs.offcanvas', function () {
			if (!scrollTarget) return;
			scrollTarget.scrollTop = 0;
			updateScrollProgress();

function fmtDate(iso) {
	if (!iso) return "";
	var s = String(iso).trim();
	if (!s) return "";
	var d = new Date(s.slice(0, 10) + "T00:00:00");
	if (Number.isNaN(d.getTime())) return s;
	return d.toLocaleDateString(undefined, { year: "numeric", month: "short", day: "numeric" });
}

function statusClass(status) {
	var s = String(status || "").trim().toLowerCase();
	if (s === "open") return "text-bg-success";
	if (s === "closed") return "text-bg-danger";
	if (s === "awarded") return "text-bg-primary";
	if (s === "canceled" || s === "cancelled") return "text-bg-dark";
	if (s === "pending") return "text-bg-secondary";
	return "text-bg-secondary";
}
