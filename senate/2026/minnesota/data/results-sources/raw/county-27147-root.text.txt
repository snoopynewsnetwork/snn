document.addEventListener('DOMContentLoaded', function () {
	const $ = jQuery;

	const $input = $('#autocompleteme');
	const $spinner = $('#autocomplete-spinner');
	const $results = $('#searchResults');
	const $clearBtn = $('.clear-search');
	const $noResults = $('.no-instant-results');
	const $searchButtonsWrap = $('.all-search-buttons-wrap');
	const $searchPopoverText = $('.searchPopover-text');
	const $searchHelp = $('.search-help');
	const $popularPages = $('.popular-pages-search');
	const modal = document.getElementById('searchModal');

	$.widget("custom.noAutoCloseAutocomplete", $.ui.autocomplete, {
		_close: function () {},
		_renderItem: function (ul, item) {
			return $("<li>")
				.append("<div>" + item.label + "</div>")
				.appendTo(ul);
		}
	});

	const allPages = [];
	
	async function preloadSitemap() {
		if (allPages.length) { return; }
		
		fetch('_assets_/sitemap.xml').then(async function(res) {
			if (!res.ok) { return; }
			
			var xmlString = await res.text();
			
			var xmlParser = new DOMParser();
			var resDoc = xmlParser.parseFromString(xmlString, "application/xml");
			
			resDoc.querySelectorAll("url").forEach(function(el) {
				
				var lab = el.querySelector("title").textContent;
				var val = el.querySelector("loc").textContent;
				var link = el.querySelector("loc").textContent;
				
				allPages.push({label: lab, value: val, link: link});
			});
		});
	}

	function updateSearchUI(val) {
		if (val.length > 0) {
			$('.all-search-buttons .btn .search-query-text').text(val);
			if (!$searchButtonsWrap.is(':visible')) {
				$searchButtonsWrap.stop(true, true).slideDown(300);
			}
			$('.all-site-search-btn').attr('href', 'search.php?q=' + encodeURIComponent(val));
			$('.all-doc-search-btn').attr('href', 'search.php?q=' + encodeURIComponent('filetype:doc OR filetype:pdf OR filetype:docx ' + val));
			$popularPages.addClass('border-top');
		} else {
			$searchButtonsWrap.stop(true, true).slideUp(300);
			$searchHelp.hide(300);
			$noResults.hide(300);
			$popularPages.removeClass('border-top');
			$searchPopoverText.text("Or try searching...");
			$results.removeClass('open').hide(300);
		}
	}

	function initAutocomplete() {
		$input.noAutoCloseAutocomplete({
			appendTo: "#searchResults",
			minLength: 1,
			delay: 300,
			html: false,
			source: function (request, response) {
				const term = $.trim(request.term).toLowerCase();
				const words = term.split(/\s+/).filter(Boolean);

				$noResults.hide(300);
				$spinner.show();

				const matches = allPages.filter(item => {
					const value = item.value.toLowerCase();
					return words.every(w => value.includes(w));
				});

				// No highlighting â return label exactly as-is
				const cleaned = matches.map(item => ({
					label: item.label,
					value: item.value,
					link: item.link
				}));

				setTimeout(() => {
					$spinner.fadeOut(1000);

					if (!term || cleaned.length === 0) {
						$input.noAutoCloseAutocomplete('close');
						$('#searchResults ul').empty();
						$noResults.show(300);
						$searchHelp.hide(300);
						$results.removeClass('open').hide(300);
						$searchPopoverText.text("Or try searching...");
					} else {
						$noResults.hide(300);
						$searchHelp.show(300);
						$results.addClass('open').show(300);
						$searchPopoverText.text("If none of the results above are helpful, you can try searching...");
					}

					response(cleaned);
				}, 300);
			},
			select: function (event, ui) {
				window.location.href = ui.item.link;
			},
			open: function () {},
			close: function () {
				$results.removeClass('open').hide(300);
				$searchHelp.hide(300);
				$searchPopoverText.text("Or try searching...");
			}
		});
	}

	initAutocomplete();

	modal.addEventListener('shown.bs.modal', function () {
		preloadSitemap();
		
		$input.val('');
		$input.trigger('input');
		$input.noAutoCloseAutocomplete('close');
		$input.trigger('focus');
	});

	$clearBtn.on('click', function (e) {
		e.preventDefault();
		$input.val('');
		$input.trigger('input');
		$input.noAutoCloseAutocomplete('close');
		$noResults.hide(300);
		$searchHelp.hide(300);
		$results.removeClass('open').hide(300);
		if ($input.data("ui-autocomplete")) {
			$input.data("ui-autocomplete").term = null;
		}
		$input.trigger('focus');
	});

	$input.on('input', function () {
		updateSearchUI($input.val());
	});

function clickLanguage(langText, maxTries, intervalMs) {
	langText = (langText || '').trim();
	maxTries = maxTries || 30;
	intervalMs = intervalMs || 200;
	var tries = 0;
	var timer = setInterval(function () {
		var $frame = $("iframe.skiptranslate");
		if ($frame.length) {
			var doc = $frame.contents();
			var $target = doc.find('span.text').filter(function () {
				return $(this).text().trim() === langText;
			}).closest('a');
			if (!$target.length) {
				$target = doc.find('span.text').filter(function () {
					return $(this).text().trim() === 'English';
				}).closest('a');
			}
			if ($target.length) {
				$target.get(0).click();
				clearInterval(timer);
				return;
			}
		}
		tries++;
		if (tries >= maxTries) clearInterval(timer);
	}, intervalMs);
}

function getCookie(cname) {
	let name = cname + "=";
	let ca = document.cookie.split(';');
	for(let i = 0; i < ca.length; i++) {
		let c = ca[i];
		while (c.charAt(0) == ' ') c = c.substring(1);
		if (c.indexOf(name) == 0) return c.substring(name.length, c.length);
	}
	return "";
}

function sanitizeFlag(code) {
	var v = (code || "").toLowerCase().trim();
	if (!v || v === "undefined" || v === "null") return "";
	if (!/^[a-z]{2}$/.test(v)) return "";
	return v;
}

function applyActiveFromCookie() {
	var code = sanitizeFlag(getCookie('lang')) || sanitizeFlag(getCookie('code')) || 'us';
	var $btn = $('.translation-links button').filter(function(){
		var dataCode = ($(this).data('code') || '').toString().toLowerCase();
		if (dataCode && dataCode === code) return true;
		var cls = ($(this).find('span[class*="fi-"]').attr('class') || '');
		var m = cls.match(/\bfi-([a-z]{2})\b/i);
		return m && m[1].toLowerCase() === code;
	}).first();
	if (!$btn.length && code === 'us') $btn = $('.translation-links button[data-lang="English"]').first();
	$('.translation-links button').removeClass('active');
	if ($btn.length) $btn.addClass('active');
}

		var q = norm($input.val());

	}

	function indexByIdOrRid(data){
		if (!data) return {};
		if (!Array.isArray(data)) return data;
		return data.reduce((m, item) => {
			if (!item) return m;
			const key = item.rid ?? item.id;
			if (key != null) m[String(key)] = item;
			return m;
		}, {});
	}

function openCatalogueDetails(rid) {
	if (!catalogueDetailsMap.modal || !catalogueJSONData) return;

	rid = String(rid);
	var cRecord = catalogueJSONData[rid];
	if (!cRecord) return;

	var m = catalogueDetailsMap;

	var parsedOptions = cRecord.options;
	if (typeof parsedOptions === 'string') {
		try { parsedOptions = JSON.parse(parsedOptions); } catch(e){ parsedOptions = {}; }
	}
	parsedOptions = parsedOptions || {};

	var imgURL = (cRecord.image || '').trim();
	if (imgURL && !imgURL.endsWith('/')) {
		m.img && m.img.classList.remove('d-none');
		if (m.img) {
			m.img.style.backgroundImage = "url('" + imgURL + "')";
		}
	} else {
		if (m.img) {
			m.img.style.backgroundImage = '';
			m.img.classList.add('d-none');
		}
	}

	// Title
	if (m.title) {
		var rawTitle = cRecord.title || cRecord.name || "";
		m.title.textContent = decodeHtmlEntities(rawTitle);
	}

	if (m.helpful) {
		var links = m.helpful.querySelectorAll("a");
		if (links.length) {
			var el = links[0];
			var webUrl  = (parsedOptions.website_url || "").trim();
			var webText = (parsedOptions.website_text || "Visit Website").trim();
			if (webUrl && webUrl !== "#") {
				el.href = webUrl;
				el.target = "_blank";
				var span = el.querySelector("span");
				if (span) span.innerText = webText;
				el.parentElement && el.parentElement.classList.remove("d-none");
			} else {
				el.href = "";
				el.parentElement && el.parentElement.classList.add("d-none");
			}
		}
	}

	if (m.maplink) {
		var mapAs = m.maplink.querySelectorAll("a");
		if (mapAs.length) {
			var mapEl = mapAs[0];
			var addr = (parsedOptions.val_address || "").trim();
			if (addr) {
				var encoded = encodeURIComponent(addr).replace(/%20/g, "+");
				mapEl.href = "https://www.google.com/maps/search/?api=1&query=" + encoded;
				mapEl.target = "_blank";
				mapEl.parentElement && mapEl.parentElement.classList.remove('d-none');
			} else {
				mapEl.href = "";
				mapEl.parentElement && mapEl.parentElement.classList.add('d-none');
			}
		}
	}

	if (m.contacts) {
		m.contacts.innerHTML = "";
		if (Array.isArray(parsedOptions.contacts)) {
			parsedOptions.contacts.forEach(function(contact) {
				var newAddress = document.createElement("address");
				newAddress.innerHTML = '<strong class="small text-muted">' + (contact.val_header||'') + '</strong><br>';

				var inner = contact.val_text || '';
				if (contact.val_href) {
					try {
						var href = contact.val_href;
						var isSelf = href.startsWith('tel:') || href.startsWith('mailto:') || href.startsWith(location.origin);
						inner = '<a href="'+href+'" target="'+(isSelf?'_self':'_blank')+'">'+ inner +'</a>';
					} catch(e) { /* ignore bad href */ }
				}
				newAddress.innerHTML += inner;
				m.contacts.append(newAddress);
			});
		}
	}

	if (m.hours) {
		const table = m.hours; 
		const dayKeys = [
			'val_sunday','val_monday','val_tuesday','val_wednesday',
			'val_thursday','val_friday','val_saturday'
		];
	
		const tds = Array.from(table.querySelectorAll('tbody td'));
		let hasAny = false;
	
		tds.forEach((td, i) => {
			const val = parsedOptions.hours && parsedOptions.hours[dayKeys[i]];
			const v = (val || '').trim();
			td.textContent = v;
			if (v) hasAny = true;
		});
	
		table.classList.toggle('d-none', !hasAny);
	}

	if (m.detail) {
		const pane = m.detail;
		const tabPane = pane.closest('.tab-pane');
		const tabLink = tabPane ? document.querySelector(`[data-bs-target="#${tabPane.id}"]`) : null;
	
		pane.innerHTML = "Loading...";
	
		fetch('catalogue/' + encodeURIComponent(rid) + '.php?display=json')
			.then(res => res.json())
			.then(data => {
				const html = (data && data.details) ? data.details.trim() : "";
	
				pane.innerHTML = html || "";
	
				const isEmpty = !html;
				if (tabPane) tabPane.classList.toggle('d-none', isEmpty);
				if (tabLink) tabLink.classList.toggle('d-none', isEmpty);
			})
			.catch(() => {
				pane.innerHTML = "";
				if (tabPane) tabPane.classList.add('d-none');
				if (tabLink) tabLink.classList.add('d-none');
			});
	}

	// Gallery #TODO)
	if (m.gallery) m.gallery.innerHTML = "";

	try {
		bootstrap.Offcanvas.getOrCreateInstance(m.modal).show();
	} catch(e) {
		console.error('Offcanvas show failed:', e);
	}

	if (window.pushOcState) pushOcState('catalogueDetails', rid);
}

function hasImg(url) {
	if (!url) return false;
	var s = String(url).trim();
	if (!s) return false;
	if (/^(null|undefined|none|n\/a)$/i.test(s)) return false;

}

		var current   = scrollTarget.scrollTop || 0;
	
		var pct = 0;
		if (maxScroll > 0) {
			pct = (current / maxScroll) * 100;
		}
		pct = Math.min(Math.max(pct, 0), 100);
	
		progressBar.style.width = pct + '%';
		if (progressOuter) {
			progressOuter.setAttribute('aria-valuenow', Math.round(pct));
		}
	}
	
	if (scrollTarget && progressBar) {
		scrollTarget.addEventListener('scroll', updateScrollProgress, { passive: true });
	
		modal.addEventListener('show.bs.offcanvas', function () {
			if (!scrollTarget) return;
			scrollTarget.scrollTop = 0;
			updateScrollProgress();

function fmtDate(iso) {
	if (!iso) return "";
	var s = String(iso).trim();
	if (!s) return "";
	var d = new Date(s.slice(0, 10) + "T00:00:00");
	if (Number.isNaN(d.getTime())) return s;
	return d.toLocaleDateString(undefined, { year: "numeric", month: "short", day: "numeric" });
}

function statusClass(status) {
	var s = String(status || "").trim().toLowerCase();
	if (s === "open") return "text-bg-success";
	if (s === "closed") return "text-bg-danger";
	if (s === "awarded") return "text-bg-primary";
	if (s === "canceled" || s === "cancelled") return "text-bg-dark";
	if (s === "pending") return "text-bg-secondary";
	return "text-bg-secondary";
}

$(document).ready(function() {
    function loadCalendarData() {
        const footer = $('.site-footer');
        if (footer.length === 0) {
            setTimeout(loadCalendarData, 500);
            return;
        }
        const calendarItems = $('#calendar-filters li:not(.calendar-tab-filter-all)');
        
        if (calendarItems.length > 0) {
            let calendarData = [];
            calendarItems.each(function() {
                const name = $(this).find('.calendar-name-filter').text().trim();
                const bgColor = $(this).css('backgroundColor');
                
                if (name !== 'Master' && name.toLowerCase() !== 'master') {
                    calendarData.push({ name, bgColor });
                }
            });
            
            if (calendarData.length > 0) {
                localStorage.setItem('calendarNames', JSON.stringify(calendarData));
                renderData(footer, calendarData);
            }
        } else {
            let calendarData = localStorage.getItem('calendarNames');
            if (calendarData) {
                calendarData = JSON.parse(calendarData);
                renderData(footer, calendarData);
            }
        }
    }
    
    function renderData(footer, calendarData) {
        footer.find('.calendar-names').remove();
        const calendarNamesDiv = $('<div class="calendar-names"></div>');
        
        const dropdown = $('#calDropdownMenu');
        if (dropdown.length > 0) {
            dropdown.empty();
        }
        
        calendarData.forEach(function(item) {
            calendarNamesDiv.append($('<span></span>').text(item.name));
            
            if (dropdown.length > 0) {
                dropdown.append($('<li></li>')
                    .text(item.name)
                    .attr('data-id', item.name.toLowerCase().replace(/ /g, ''))
                );
            }
        });
        
        footer.append(calendarNamesDiv);
    }
    
    const observer = new MutationObserver(function() {
        loadCalendarData();
    });
    
    const calendarFilters = document.getElementById('calendar-filters');
    if (calendarFilters) {
        observer.observe(calendarFilters, {
            childList: true,
            subtree: true,
            characterData: true
        });
    }
    
    loadCalendarData();
});
