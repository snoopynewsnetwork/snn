        $(document).ready(function() {
        var prompt = "Search Here";
        var $s = $(".search.bar .searchField");
        $s.focus(clearPrompt).blur(putPrompt).change(trimfield);
        function clearPrompt() { if ($s.val() == prompt) $s.val(""); }

        $('#searchField').val(searchval);
        }
        putPrompt();

					var resp =
						{ 0: { items: 1 },
						600: { items: 2 },

						1000: { items: 4 },

							if (qlMaxItems < 9) {
								$.each(resp, function(index, value) {
									if (value.items > qlMaxItems) { delete resp[index]; }
								});
							}
							if (qlMaxItems > qlcount) {
								$.each(resp, function(index, value) {
									if (value.items > qlcount) { delete resp[index]; }
								});
							}
							function SetupQuickLinks() {
								let numDisplayed = 0;
								$.each(resp, function(index, value) {

									owl.find('.owl-item').contents().unwrap();
									owl.owlCarousel({ center: false, loop: false, autoplay: qlautoplay, autoHeight: false,
									autoplayTimeout: 8000, autoplayHoverPause: true, smartSpeed: 250, dots: false,
									nav: false, responsive: resp
									});
									$(controls).css('display', 'none');
