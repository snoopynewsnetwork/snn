    (function() {
        var wrapper = document.getElementById('ocsoe-cpt-ajax-30615');

        if (!wrapper) {
            return;
        }

        wrapper.addEventListener('click', function(event) {
            var link = event.target.closest('.ocsoe-cpt-page-link');

            if (!link) {
                return;
            }

            event.preventDefault();

            var page = link.getAttribute('data-page');
            var listWrap = wrapper.querySelector('.ocsoe-cpt-ajax-list');
            var paginationWrap = wrapper.querySelector('.ocsoe-cpt-ajax-pagination-wrap');

            var formData = new FormData();

            formData.append('action', 'ocsoe_load_cpt_ajax_items');
            formData.append('nonce', wrapper.getAttribute('data-nonce'));
            formData.append('page', page);
            formData.append('post_type', wrapper.getAttribute('data-post-type'));
            formData.append('taxonomy', wrapper.getAttribute('data-taxonomy') || '');
            formData.append('term', wrapper.getAttribute('data-term') || '');
            formData.append('limit', wrapper.getAttribute('data-limit') || 6);
            formData.append('order', wrapper.getAttribute('data-order') || 'DESC');
            formData.append('orderby', wrapper.getAttribute('data-orderby') || 'date');
            formData.append('button_text', wrapper.getAttribute('data-button-text') || 'READ MORE');

            wrapper.classList.add('is-loading');

            fetch(wrapper.getAttribute('data-ajax-url'), {
                method: 'POST',
                body: formData
            })
            .then(function(response) {
                return response.json();
            })
            .then(function(response) {
                if (!response.success) {
                    wrapper.classList.remove('is-loading');
                    return;
                }

                listWrap.innerHTML = response.data.items;
                paginationWrap.innerHTML = response.data.pagination;

                wrapper.classList.remove('is-loading');

                wrapper.scrollIntoView({
                    behavior: 'smooth',
                    block: 'start'
                });
            })
            .catch(function() {
                wrapper.classList.remove('is-loading');
            });
        });
    })();
