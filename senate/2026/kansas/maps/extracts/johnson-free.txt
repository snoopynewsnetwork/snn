        document.addEventListener('DOMContentLoaded', () => {

            const menuToggle = document.getElementById('menu-toggle');
            const mainMenu = document.getElementById('main-menu');

            /* ---------- Utilities ---------- */

            function openMenu(button, menu) {
                //closeOpenMenus(button, menu);   //dsteen added

                button.setAttribute('aria-expanded', 'true');
                menu.hidden = false;

                const firstItem = menu.querySelector('a, button');
                firstItem && firstItem.focus();
            }

            function closeOpenMenus(button, menu) {    //dsteen added
                document.querySelectorAll('.site-nav button[aria-expanded="true"]')
                    .forEach(button => {
                        const menu = document.getElementById(
                            button.getAttribute('aria-controls')
                        );
                        if (
                            menu //&&
                            //!button.contains(e.target) &&
                            //!menu.contains(e.target)
                        ) {
                            button.setAttribute('aria-expanded', 'false');
                            menu.hidden = true;
                        }
                    });
            }


            function closeMenu(button, menu) {
                button.setAttribute('aria-expanded', 'false');
                menu.hidden = true;
                button.focus();
            }

            function toggleMenu(button, menu) {
                const expanded = button.getAttribute('aria-expanded') === 'true';
                if (expanded) {
                    closeMenu(button, menu);
                } else {
                    openMenu(button, menu);
                }
            }

            /* ---------- Hamburger (mobile) ---------- */

            if (menuToggle && mainMenu) {
                // 2025-04-10 - dsteen - don't need to add this here because it gets added below in the document.forEach(button => {...
                //menuToggle.addEventListener('click', e => {
                //    e.stopPropagation();
                //    event.preventDefault();
                //    toggleMenu(menuToggle, mainMenu);
                //});

                mainMenu.addEventListener('keydown', e => {
                    if (e.key === 'Escape') {
                        e.stopPropagation();
                        closeMenu(menuToggle, mainMenu);
                        event.preventDefault(); //2026-04-10 - dsteen - added to prevent page postback
                    }
                });
            }

            /* ---------- Submenus & Flyouts ---------- */

            document
                .querySelectorAll('.site-nav button[aria-controls]')
                .forEach(button => {
                    const menuId = button.getAttribute('aria-controls');
                    const menu = document.getElementById(menuId);

                    if (!menu) return;

                    /* Click opens/closes */
                    button.addEventListener('click', e => {
                        e.stopPropagation();
                        toggleMenu(button, menu);
                        event.preventDefault();  //2026-04-10 - dsteen - added to prevent page postback
                    });

                    /* Escape on the button closes its own menu */
                    button.addEventListener('keydown', e => {
                        if (e.key === 'Escape') {
                            e.stopPropagation();
                            closeMenu(button, menu);
                        }
                    });

                    /* Escape anywhere inside the menu closes ONLY this level */
                    menu.addEventListener('keydown', e => {
                        if (e.key === 'Escape') {
                            e.stopPropagation();
                            closeMenu(button, menu);
                        }
                    });
                });

            /* ---------- Click outside closes open menus ---------- */

            document.addEventListener('click', e => {
                document
                    .querySelectorAll('.site-nav button[aria-expanded="true"]')
                    .forEach(button => {
                        const menu = document.getElementById(
                            button.getAttribute('aria-controls')
                        );
                        console.log(menu);
                        if (
                            menu &&
                            !button.contains(e.target) &&
                            !menu.contains(e.target)
                        ) {
                            button.setAttribute('aria-expanded', 'false');
                            menu.hidden = true;
                        }
                    });
            });

        });

     Free Data 
   
   
    AIMS offers the datasets listed on this page at no cost. Most datasets use  State Plane North  projection.
   
   
    In addition, an ArcGIS Server Service containing each dataset below is available via our ArcGIS Server Services  Portal .
   
   
    If a dataset you are interested in is not listed, it may be available to purchase from our  Digital Data Request  application.
   
   
    Please refer to the  GIS Data Listing  page for metadata.
